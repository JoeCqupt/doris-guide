# Download raw query profiles for the actually profiled sessions.
headers={'Authorization':'Basic '+base64.b64encode(b'root:').decode()}
def get(url):
 with urllib.request.urlopen(urllib.request.Request(url,headers=headers),timeout=20) as f:return f.read()
raw=get('http://127.0.0.1:8030/rest/v1/query_profile');(r/'profiles-index.raw.json').write_bytes(raw)
profiles=r/'profiles';profiles.mkdir(exist_ok=True)
for row in json.loads(raw)['data']['rows']:
 ident=row['Profile ID']
 if row['Task Type']!='QUERY':continue
 data=get('http://127.0.0.1:8030/rest/v1/query_profile/text/'+ident)
 (profiles/(ident+'.json')).write_bytes(data)
 value=json.loads(data)['data'];(profiles/(ident+'.txt')).write_text(value if isinstance(value,str) else json.dumps(value,ensure_ascii=False,indent=2))
print('Profiles downloaded',len(list(profiles.glob('*.txt'))),flush=True)
