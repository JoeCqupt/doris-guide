from pathlib import Path
import subprocess,json,datetime
R=Path('/Users/jiangyuan/doris-guide/doris-docs/examples/07-live-lab-20261014')
MYSQL=['/opt/homebrew/bin/mysql','--connect-timeout=5','--default-character-set=utf8mb4','-h127.0.0.1','-P9030','-uroot','--batch','--raw']
records=[]
def run(name,sql,expected=None,fail=False):
 (R/(name+'.sql')).write_text(sql+'\n')
 p=subprocess.run(MYSQL+(['--skip-column-names'] if expected is not None else ['-vvv']),input=sql,text=True,capture_output=True,timeout=120)
 (R/(name+'.out.txt')).write_text(p.stdout)
 (R/(name+'.stderr.txt')).write_text(p.stderr)
 ok=(p.returncode!=0) if fail else p.returncode==0
 if expected is not None:ok=ok and p.stdout.strip()==expected
 row={'case':name,'exit_code':p.returncode,'passed':ok,'expected_error':fail,'expected_output':expected,'time':datetime.datetime.now(datetime.timezone.utc).isoformat()};records.append(row)
 (R/'results.json').write_text(json.dumps(records,ensure_ascii=False,indent=2)+'\n')
 print(name,'PASS' if ok else 'FAIL',flush=True)
 if not ok:print((p.stderr+p.stdout)[-1300:],flush=True)
 return ok
